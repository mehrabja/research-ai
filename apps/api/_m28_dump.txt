from __future__ import annotations

import uuid
from dataclasses import dataclass
from typing import Literal

RetrievalMode = Literal["semantic", "lexical", "hybrid"]
RetrievalAlgorithm = Literal["exact", "ann"]

RETRIEVAL_ALGORITHM_VERSION: str = "m9.1"


@dataclass(frozen=True)
class RetrievedChunk:
    chunk_id: uuid.UUID
    source_id: uuid.UUID
    report_id: uuid.UUID
    chunk_index: int
    content: str
    char_start: int | None
    char_end: int | None
    page_number: int | None

    vector_distance: float | None
    vector_similarity: float | None
    lexical_score: float | None

    vector_rank: int | None
    lexical_rank: int | None
    fusion_score: float
    final_rank: int

    def evidence_id(self) -> str:
        return f"source:{self.source_id}:chunk:{self.chunk_id}"


def reciprocal_rank_fusion_score(
    *,
    vector_rank: int | None,
    lexical_rank: int | None,
    rrf_k: int,
    vector_weight: float,
    lexical_weight: float,
) -> float:
    if rrf_k <= 0:
        raise ValueError("rrf_k must be > 0")
    if vector_weight < 0 or lexical_weight < 0:
        raise ValueError("weights must be >= 0")
    if vector_weight + lexical_weight <= 0:
        raise ValueError("at least one weight must be > 0")

    score = 0.0
    if vector_rank is not None and vector_rank >= 1:
        score += vector_weight * (1.0 / (rrf_k + vector_rank))
    if lexical_rank is not None and lexical_rank >= 1:
        score += lexical_weight * (1.0 / (rrf_k + lexical_rank))
    return score

def apply_plan_filters(
    candidates: list[Candidate],
    *,
    min_year: int | None,
    max_year: int | None,
    required_full_text: bool,
    sources_by_id: dict,
) -> tuple[list[Candidate], list[str]]:
    warnings: list[str] = []
    if not (min_year or max_year or required_full_text):
        return candidates, warnings

    kept: list[Candidate] = []
    dropped_year = 0
    dropped_full_text = 0

    for c in candidates:
        src = sources_by_id.get(c.source_id)
        if src is None:
            kept.append(c)
            continue
        if min_year is not None and (src.year or 0) < min_year:
            dropped_year += 1
            continue
        if max_year is not None and (src.year or 9999) > max_year:
            dropped_year += 1
            continue
        if (
            required_full_text
            and not _is_full_text_available(src)
        ):
            dropped_full_text += 1
            continue
        kept.append(c)

    if dropped_year:
        warnings.append(f"filtered_by_year: {dropped_year}")
    if dropped_full_text:
        warnings.append(f"filtered_by_full_text: {dropped_full_text}")
    return kept, warnings


def apply_requirement_filter(
    candidates: list[Candidate],
    *,
    requirement: ResearchEvidenceRequirement | None,
    sources_by_id: dict[uuid.UUID, Source],
    literature_by_source_id: dict[
        uuid.UUID, ProjectLiteratureItem
    ],
) -> PlanFilterResult:
    """Apply M24 requirement constraints to candidates.

    Candidate-level filters only. ``min_sources`` is a threshold
    outcome, not a candidate filter â€” it never deletes candidates.
    """
    warnings: list[str] = []
    rejected: dict[str, int] = {
        "paper_type": 0,
        "research_design": 0,
        "max_age": 0,
        "full_text": 0,
        "min_relevance": 0,
        "screening_status": 0,
    }

    if requirement is None:
        distinct = count_distinct_sources(candidates)
        return PlanFilterResult(
            candidates=candidates,
            warnings=warnings,
            rejected_counts={},
            distinct_source_count=distinct,
            min_sources_satisfied=True,
        )

    preferred_pt = requirement.preferred_paper_types
    preferred_rd = requirement.preferred_research_designs
    max_age = requirement.max_age_years
    min_rel = requirement.min_relevance_score
    full_text_required = bool(requirement.full_text_required)

    kept: list[Candidate] = []
    for c in candidates:
        src = sources_by_id.get(c.source_id)
        lit = literature_by_source_id.get(c.source_id)

        if (
            lit is not None
            and lit.screening_status not in _SCREENING_INCLUDED
        ):
            rejected["screening_status"] += 1
            continue

        if preferred_pt and (
            lit is None
            or not _matches_code_set(lit.paper_type_code, preferred_pt)
        ):
            rejected["paper_type"] += 1
            continue

        if preferred_rd and (
            lit is None
            or not _matches_code_set(
                lit.research_design_code, preferred_rd
            )
        ):
            rejected["research_design"] += 1
            continue

        if min_rel is not None:
            if lit is None or lit.relevance_score is None:
                rejected["min_relevance"] += 1
                continue
            if float(lit.relevance_score) < float(min_rel):
                rejected["min_relevance"] += 1
                continue

        if max_age is not None:
            if src is None or src.year is None:
                rejected["max_age"] += 1
                continue
            if (
                RESEARCH_PLAN_REFERENCE_YEAR - src.year
            ) > max_age:
                rejected["max_age"] += 1
                continue

        if (
            full_text_required
            and not _is_full_text_available(src)
        ):
            rejected["full_text"] += 1
            continue

        kept.append(c)

    for key, count in rejected.items():
        if count > 0:
            warnings.append(f"requirement_filtered_{key}: {count}")

    distinct = count_distinct_sources(kept)
    min_sources = requirement.min_sources
    min_satisfied = True
    if (
        min_sources is not None
        and min_sources > 0
        and distinct < min_sources
    ):
        min_satisfied = False
        warnings.append(
            "requirement_min_sources_unsatisfied: "
            f"required={min_sources} got={distinct}"
        )

    rejected_nonzero = {
        k: v for k, v in rejected.items() if v > 0
    }

    return PlanFilterResult(
        candidates=kept,
        warnings=warnings,
        rejected_counts=rejected_nonzero,
        distinct_source_count=distinct,
        min_sources_satisfied=min_satisfied,
    )

"""M26 â€” Hybrid scoring + deterministic ranking."""
from __future__ import annotations

from dataclasses import dataclass, field
from typing import Any

from app.db.models.source import Source
from app.services.rag.candidates import Candidate


@dataclass
class ScoredCandidate:
    candidate: Candidate
    final_score: float
    vector_score: float | None = None
    lexical_score: float | None = None
    graph_score: float | None = None
    memory_score: float | None = None
    quality_score: float | None = None
    diversity_penalty: float | None = None
    reasons: list[str] = field(default_factory=list)
    graph_distance: int | None = None
    memory_match: str | None = None
    meta: dict[str, Any] = field(default_factory=dict)


def _clamp(v: float) -> float:
    return max(0.0, min(1.0, v))


def quality_score_for_source(source: Source | None) -> float | None:
    if source is None:
        return None
    if source.source_quality is not None:
        try:
            return _clamp(float(source.source_quality))
        except (TypeError, ValueError):
            return None
    status = (source.status or "").lower()
    if status in ("structured_full_text", "full_text"):
        return 0.85
    if status == "abstract_only":
        return 0.55
    if status == "metadata_only":
        return 0.35
    if status == "extraction_failed":
        return 0.15
    return None


def score_candidates(
    candidates: list[Candidate],
    *,
    sources_by_id: dict,
) -> list[ScoredCandidate]:
    out: list[ScoredCandidate] = []
    for c in candidates:
        src = sources_by_id.get(c.source_id)
        q = quality_score_for_source(src)

        v_sim = (
            1.0 - c.vector_distance
            if c.vector_distance is not None else None
        )
        if v_sim is not None:
            v_sim = _clamp(v_sim)

        l_norm = (
            _clamp(c.lexical_score / 2.0)
            if c.lexical_score is not None else None
        )

        g_score: float | None = None
        g_dist: int | None = None
        if c.stage.startswith("graph:d"):
            try:
                g_dist = int(c.stage.split("d", 1)[1])
                g_score = _clamp(1.0 / (1.0 + g_dist))
            except (ValueError, IndexError):
                g_score = None

        base_parts: list[float] = [c.fusion_score]
        reasons: list[str] = ["hybrid_fusion"]
        if v_sim is not None:
            base_parts.append(v_sim)
            reasons.append("vector_match")
        if l_norm is not None:
            base_parts.append(l_norm)
            reasons.append("lexical_match")
        if g_score is not None:
            base_parts.append(g_score)
            reasons.append("graph_neighbor")
        if q is not None:
            base_parts.append(q)
            reasons.append("source_quality")

        final = sum(base_parts) / max(1, len(base_parts))

        out.append(ScoredCandidate(
            candidate=c,
            final_score=final,
            vector_score=v_sim,
            lexical_score=l_norm,
            graph_score=g_score,
            memory_score=None,
            quality_score=q,
            reasons=reasons,
            graph_distance=g_dist,
        ))
    out.sort(key=lambda s: (-s.final_score, str(s.candidate.chunk_id)))
    return out

    "score_candidates",
    "quality_score_for_source",

===== FILE: app\services\rag\diversity.py =====
"""M26 â€” MMR-style diversity with deterministic tie-breaking."""
from __future__ import annotations

from app.services.rag.scoring import ScoredCandidate


def _source_key(s: ScoredCandidate) -> str:
    return str(s.candidate.source_id)


def _chunk_key(s: ScoredCandidate) -> str:
    return str(s.candidate.chunk_id)


def apply_mmr(
    scored: list[ScoredCandidate],
    *,
    lambda_: float,
    max_results: int,
) -> list[ScoredCandidate]:
    if not scored or max_results <= 0:
        return []
    lambda_ = max(0.0, min(1.0, lambda_))
    selected: list[ScoredCandidate] = []
    remaining = list(scored)

    remaining.sort(key=lambda s: (-s.final_score, _chunk_key(s)))
    selected.append(remaining.pop(0))

    while remaining and len(selected) < max_results:
        best_idx = 0
        best_value = float("-inf")
        best_tie = ""
        selected_sources = {_source_key(s) for s in selected}

        for i, cand in enumerate(remaining):
            redundancy = 0.0
            if _source_key(cand) in selected_sources:
                redundancy = 1.0
            mmr = (
                lambda_ * cand.final_score
                - (1.0 - lambda_) * redundancy
            )
            tie = _chunk_key(cand)
            if mmr > best_value or (mmr == best_value and tie < best_tie):
                best_value = mmr
                best_idx = i
                best_tie = tie

        chosen = remaining.pop(best_idx)
        chosen.diversity_penalty = (
            0.0 if _source_key(chosen) not in selected_sources else 1.0
        )
        if chosen.diversity_penalty == 0.0:
            chosen.reasons.append("diversity_selected")
        selected.append(chosen)

    return selected


__all__ = ["apply_mmr"]

def load_gap_source_ids(
    db: Session,
    *,
    project_id: uuid.UUID,
    graph_version_id: uuid.UUID,
) -> set[uuid.UUID]:
    """Sources downstream of literature_gap nodes.

    Chain: literature_gap â†’ literature_cluster
           â†’ literature_item â†’ source
    """
    gap_node_ids = list(db.execute(
        select(KnowledgeGraphNode.id).where(
            KnowledgeGraphNode.graph_version_id == graph_version_id,
            KnowledgeGraphNode.project_id == project_id,
            KnowledgeGraphNode.node_type == "literature_gap",

        return set()

    cluster_edges = list(db.execute(
        select(KnowledgeGraphEdge.target_node_id).where(
            KnowledgeGraphEdge.graph_version_id == graph_version_id,
            KnowledgeGraphEdge.source_node_id.in_(gap_node_ids),
            KnowledgeGraphEdge.relation_type
            == "gap_detected_from_cluster",
        )
    ).scalars().all())
    if not cluster_edges:
        return set()

    item_edges = list(db.execute(
        select(KnowledgeGraphEdge.target_node_id).where(
            KnowledgeGraphEdge.graph_version_id == graph_version_id,
            KnowledgeGraphEdge.source_node_id.in_(cluster_edges),
            KnowledgeGraphEdge.relation_type
            == "cluster_contains_literature_item",
        )
    ).scalars().all())
    if not item_edges:
        return set()

    source_edges = list(db.execute(
        select(KnowledgeGraphEdge.target_node_id).where(
            KnowledgeGraphEdge.graph_version_id == graph_version_id,
            KnowledgeGraphEdge.source_node_id.in_(item_edges),
            KnowledgeGraphEdge.relation_type
            == "literature_item_refers_to_source",
        )
    ).scalars().all())
    if not source_edges:
        return set()

    return _resolve_source_entities(

) -> dict | None:

def _select_reqs_for_plan(
    db: Session, *, plan: ResearchPlan
) -> tuple[
    list[ResearchPlanSection],
    list[ResearchEvidenceRequirement],
    list[ResearchPlanDependency],
]:
    sections = list(db.execute(
        select(ResearchPlanSection)
        .where(ResearchPlanSection.plan_id == plan.id)
        .order_by(ResearchPlanSection.display_order.asc())
    ).scalars().all())
    reqs = list(db.execute(
        select(ResearchEvidenceRequirement)
        .where(ResearchEvidenceRequirement.plan_id == plan.id)
    ).scalars().all())
    deps = list(db.execute(
        select(ResearchPlanDependency)
        .where(ResearchPlanDependency.plan_id == plan.id)
    ).scalars().all())

    section_ids = {s.id for s in sections}
    for r in reqs:
        if r.section_id not in section_ids:
            raise ManuscriptInvalid(
                f"evidence requirement {r.id} references foreign section"
            )
    for d in deps:
        if d.upstream_section_id not in section_ids:
            raise ManuscriptInvalid(
                f"dependency {d.id} has foreign upstream"
            )
        if d.downstream_section_id not in section_ids:
            raise ManuscriptInvalid(
                f"dependency {d.id} has foreign downstream"
            )
    return sections, reqs, deps

def _json(value) -> str:
    return json.dumps(
        value, sort_keys=True, separators=(",", ":"), ensure_ascii=False
    )


def _sha256(parts: list[str]) -> str:
    h = hashlib.sha256()
    for p in parts:
        h.update(p.encode("utf-8"))
        h.update(b"\x00")
    return h.hexdigest()[:32]


def resolve_ready_plan(
    db: Session, *, project_id: uuid.UUID
) -> ResearchPlan | None:
    return db.execute(
        select(ResearchPlan)
        .where(
            ResearchPlan.project_id == project_id,
            ResearchPlan.status == "ready",
        )
        .order_by(ResearchPlan.version.desc())
        .limit(1)
    ).scalar_one_or_none()


def compute_manuscript_context_hash(
    db: Session,
    *,
    report: Report,

    section_keys = [
        s.section_key for s in db.execute(
            select(ResearchPlanSection)
            .where(ResearchPlanSection.plan_id == plan.id)
            .order_by(ResearchPlanSection.display_order.asc())
        ).scalars().all()
    ]

    parts: list[str] = [
        f"arch:{MANUSCRIPT_ARCHITECTURE_VERSION}",
        f"tmpl:{MANUSCRIPT_TEMPLATE_VERSION}",
        f"cov:{MANUSCRIPT_COVERAGE_MODEL_VERSION}",
        f"intent:{MANUSCRIPT_INTENT_MAPPING_VERSION}",
        f"report:{report.id}",

        + _json({

        f"plan_design:{plan.research_design_code or '-'}",

def _resolve_coverage_target(
    reqs: list[RequirementSnapshot],
    section: ResearchPlanSection,
) -> float:
    m24 = float(section.evidence_coverage_target or 0.0)
    if 0.0 < m24 <= 1.0:
        return m24
    required = [r for r in reqs if r.required]
    if not required:
        return 0.0
    weights = [r.coverage_weight for r in required if r.coverage_weight > 0]
    if not weights:
        return 0.0
    avg = sum(weights) / len(weights)
    return max(0.0, min(1.0, round(avg, 4)))


def build_section_snapshots(
    *,
    plan: ResearchPlan,
    sections: list[ResearchPlanSection],
    requirements: list[ResearchEvidenceRequirement],
    word_budget_by_section: dict[str, int] | None = None,

    for r in requirements:
        reqs_by_section.setdefault(str(r.section_id), []).append(r)

    wb = word_budget_by_section or {}
    warnings: list[str] = []
    snapshots: list[SectionSnapshot] = []

    ordered = sorted(sections, key=lambda s: (s.display_order, s.section_key))
    for s in ordered:
        sid = str(s.id)
        reqs = sorted(
            reqs_by_section.get(sid, []),
            key=lambda r: (-int(r.priority), r.requirement_key),
        )
        snap_reqs: list[RequirementSnapshot] = []
        for r in reqs:
            snap = _snapshot_requirement(r)
            if snap.required and not snap.retrieval_intents:
                warnings.append(
                    f"unmapped_required_requirement:{snap.requirement_key}"
                )
            snap_reqs.append(snap)

        target = _resolve_coverage_target(snap_reqs, s)
        snapshots.append(SectionSnapshot(
            source_plan_section_id=sid,

            purpose=s.purpose,

            budget_weight=float(s.budget_weight),

def aggregate_source_constraints(
    reqs: list[RequirementSnapshot],
) -> dict[str, Any]:
    required = [r for r in reqs if r.required]
    mins = [r.min_sources for r in required if r.min_sources is not None]
    maxs = [r.max_sources for r in required if r.max_sources is not None]
    ages = [r.max_age_years for r in required if r.max_age_years is not None]
    rels = [
        r.min_relevance_score for r in required
        if r.min_relevance_score is not None
    ]

    pref_pt: set[str] = set()
    pref_rd: set[str] = set()
    for r in reqs:
        pref_pt.update(r.preferred_paper_types)
        pref_rd.update(r.preferred_research_designs)

    return {
        "min_sources": max(mins) if mins else None,
        "max_sources": min(maxs) if maxs else None,
        "preferred_paper_types": sorted(pref_pt),
        "preferred_research_designs": sorted(pref_rd),
        "max_age_years": min(ages) if ages else None,
        "full_text_required": any(r.full_text_required for r in required),
        "min_relevance_score": max(rels) if rels else None,
    }

from __future__ import annotations

from app.db.models.manuscript import (

def topological_research_order(
    *,
    section_keys: list[str],
    edges: list[tuple[str, str]],
) -> list[str]:
    indeg = dict.fromkeys(section_keys, 0)
    adj: dict[str, list[str]] = {n: [] for n in section_keys}
    for u, v in edges:
        if u not in adj or v not in adj:
            continue
        adj[u].append(v)
        indeg[v] += 1
    queue = sorted(n for n in section_keys if indeg[n] == 0)
    out: list[str] = []
    while queue:
        u = queue.pop(0)
        out.append(u)
        for v in adj[u]:
            indeg[v] -= 1
            if indeg[v] == 0:
                queue.append(v)
                queue.sort()
    return out


def research_order_for_sections(
    *,
    sections: list[ManuscriptSection],
    dependencies: list[ManuscriptSectionDependency],
) -> list[str]:
    by_id = {s.id: s.section_key for s in sections}
    edges = [
        (by_id[d.upstream_section_id], by_id[d.downstream_section_id])
        for d in dependencies
        if d.upstream_section_id in by_id and d.downstream_section_id in by_id
    ]
    return topological_research_order(
        section_keys=[s.section_key for s in sections],
        edges=edges,
    )

@dataclass
class ManuscriptValidationResult:
    errors: list[str] = field(default_factory=list)
    warnings: list[str] = field(default_factory=list)

    def valid(self) -> bool:
        return not self.errors

    def as_dict(self) -> dict:
        return {
            "validation_status": "valid" if self.valid() else "invalid",
            "validation_errors": list(self.errors),
            "warnings": list(self.warnings),
        }


def _has_cycle(nodes: list[str], edges: list[tuple[str, str]]) -> bool:
    adj: dict[str, list[str]] = {n: [] for n in nodes}
    for u, v in edges:
        adj.setdefault(u, []).append(v)
        adj.setdefault(v, [])
    WHITE, GRAY, BLACK = 0, 1, 2
    color = dict.fromkeys(adj, WHITE)

    def dfs(u: str) -> bool:
        color[u] = GRAY
        for v in adj[u]:
            if color[v] == GRAY:
                return True
            if color[v] == WHITE and dfs(v):
                return True
        color[u] = BLACK
        return False

    return any(color[n] == WHITE and dfs(n) for n in adj)


def validate_persisted_manuscript(
    *,
    sections: list[ManuscriptSection],
    dependencies: list[ManuscriptSectionDependency],
    word_target: int,
    budget_tolerance: float = 0.10,
) -> ManuscriptValidationResult:
    res = ManuscriptValidationResult()

    if not sections:
        res.errors.append("no sections")
        return res

    keys = [s.section_key for s in sections]
    if len(keys) != len(set(keys)):
        res.errors.append("duplicate section keys")

    orders = sorted(s.display_order for s in sections)
    if orders != list(range(len(orders))):
        res.errors.append("display_order is not a continuous 0..N-1 sequence")

    total_words = 0
    for s in sections:
        if not s.title:
            res.errors.append(f"section {s.section_key}: missing title")
        if s.required and s.word_budget <= 0:
            res.errors.append(
                f"required section {s.section_key}: word_budget must be > 0"
            )
        total_words += int(s.word_budget)

        contract = s.research_contract or {}
        intents = contract.get("retrieval_intents") or []
        for i in intents:
            if i not in KNOWN_RETRIEVAL_INTENTS:
                res.errors.append(
                    f"section {s.section_key}: unknown intent {i}"
                )
        for req in contract.get("evidence_requirements") or []:
            if req.get("required"):
                req_intents = req.get("retrieval_intents") or []
                if not req_intents:
                    res.errors.append(
                        f"required requirement "
                        f"{req.get('requirement_key')} in section "
                        f"{s.section_key} has no intent mapping"
                    )
                for i in req_intents:
                    if i not in KNOWN_RETRIEVAL_INTENTS:
                        res.errors.append(
                            f"section {s.section_key}: unknown intent "
                            f"{i} in requirement "
                            f"{req.get('requirement_key')}"
                        )

    if word_target > 0:
        diff = abs(total_words - word_target) / word_target
        if diff > budget_tolerance:
            res.errors.append(
                f"budget sum {total_words} differs from word_target "
                f"{word_target} by {diff:.2%}"
            )

    by_id = {s.id: s.section_key for s in sections}
    edges: list[tuple[str, str]] = []
    seen_edges: set[tuple[str, str]] = set()
    for d in dependencies:
        up = by_id.get(d.upstream_section_id)
        down = by_id.get(d.downstream_section_id)
        if up is None or down is None:
            res.errors.append("dependency endpoint not in version")
            continue
        if up == down:
            res.errors.append(f"self-dependency: {up}")
        if (up, down) in seen_edges:
            res.errors.append(f"duplicate dependency edge: {up}->{down}")
        seen_edges.add((up, down))
        edges.append((up, down))

    if _has_cycle(keys, edges):
        res.errors.append("dependency cycle detected")

    return res
