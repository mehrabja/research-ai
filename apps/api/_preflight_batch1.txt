from __future__ import annotations

import uuid
from datetime import datetime

from sqlalchemy import DateTime, func, text
from sqlalchemy.dialects.postgresql import UUID as PGUUID
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column


class Base(DeclarativeBase):
    """Project-wide declarative base."""


class UUIDPKMixin:
    id: Mapped[uuid.UUID] = mapped_column(
        PGUUID(as_uuid=True),
        primary_key=True,
        server_default=text("gen_random_uuid()"),
    )


class CreatedAtMixin:
    created_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True),
        nullable=False,
        server_default=func.now(),
    )


class TimestampMixin(CreatedAtMixin):
    updated_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True),
        nullable=False,
        server_default=func.now(),
        onupdate=func.now(),
    )

def _in_check(col: str, values: tuple[str, ...], name: str) -> CheckConstraint:
    return CheckConstraint(
        f"{col} IN (" + ",".join(f"'{v}'" for v in values) + ")",
        name=name,
    )

"""Job service.

Owns enqueue / claim / heartbeat / complete / fail / cancel / recovery.
All state changes go through ``validate_job_transition``.

    JobPayloadTooLargeError,
)

_UNIQUE_VIOLATION = "23505"
_LEASE_MAX_MESSAGE = 500


@dataclass(frozen=True)
class EnqueueResult:
    job: Job
    created: bool


def enqueue_job(
    db: Session,
    *,
    job_type: str,
    payload: dict,
    idempotency_key: str | None = None,
    report_id: uuid.UUID | None = None,
    project_file_id: uuid.UUID | None = None,
    user_id: uuid.UUID | None = None,
    priority: int = 0,
    max_attempts: int | None = None,
    available_at: datetime | None = None,
) -> EnqueueResult:
    settings = get_settings()

    encoded = json.dumps(payload, ensure_ascii=False).encode("utf-8")
    if len(encoded) > settings.job_payload_max_bytes:
        raise JobPayloadTooLargeError()

    if report_id is not None and user_id is not None:
        get_report(db, user_id=user_id, report_id=report_id)

    effective_max = max_attempts or settings.job_default_max_attempts

    if idempotency_key is not None:
        existing = db.execute(
            select(Job).where(Job.idempotency_key == idempotency_key)
        ).scalar_one_or_none()
        if existing is not None:
            return EnqueueResult(job=existing, created=False)

    job = Job(
        job_type=job_type,
        payload=payload,
        idempotency_key=idempotency_key,
        report_id=report_id,
        user_id=user_id,
        priority=priority,
        max_attempts=effective_max,
        status="queued",
        attempt_count=0,
        available_at=available_at or datetime.now(UTC),
        project_file_id=project_file_id,
    )
    try:
        db.add(job)
        db.commit()
        db.refresh(job)
    except IntegrityError as exc:
        db.rollback()
        sqlstate = getattr(exc.orig, "sqlstate", None)
        if sqlstate == _UNIQUE_VIOLATION and idempotency_key is not None:
            winner = db.execute(
                select(Job).where(
                    Job.idempotency_key == idempotency_key
                )
            ).scalar_one()
            return EnqueueResult(job=winner, created=False)
        raise
    except Exception:
        db.rollback()
        raise
    return EnqueueResult(job=job, created=True)

def claim_job(db: Session, *, worker_id: str) -> Job | None:
    now = datetime.now(UTC)
    try:
        stmt = (
            select(Job)
            .where(
                Job.status.in_(("queued", "retrying")),
                Job.available_at <= now,
                Job.attempt_count < Job.max_attempts,
            )
            .order_by(
                Job.priority.desc(),
                Job.available_at.asc(),
                Job.created_at.asc(),
                Job.id.asc(),
            )
            .limit(1)
            .with_for_update(skip_locked=True)
        )
        job = db.execute(stmt).scalar_one_or_none()
        if job is None:
            db.rollback()
            return None

        validate_job_transition(job.status, "processing")

        job.status = "processing"
        job.attempt_count = job.attempt_count + 1
        job.worker_id = worker_id
        job.started_at = now
        job.heartbeat_at = now
        job.completed_at = None
        job.failed_at = None

        db.commit()
        db.refresh(job)
        return job
    except Exception:
        db.rollback()
        raise


def heartbeat_job(
    db: Session, *, job_id: uuid.UUID, worker_id: str
) -> bool:

    result = cast(
        CursorResult,
        db.execute(
            update(Job)
            .where(
                Job.id == job_id,
                Job.status == "processing",
                Job.worker_id == worker_id,
            )
            .values(heartbeat_at=now, updated_at=now)
        ),
    )
    if result.rowcount == 0:
        db.rollback()
        return False
    db.commit()
    return True


def complete_job(

def fail_job(
    db: Session,
    *,
    job_id: uuid.UUID,
    worker_id: str,
    error_code: str,
    error_message: str,
) -> None:
    now = datetime.now(UTC)

    job = db.execute(
        select(Job)
        .where(
            Job.id == job_id,
            Job.status == "processing",
            Job.worker_id == worker_id,
        )
        .with_for_update()
    ).scalar_one_or_none()
    if job is None:
        db.rollback()
        raise JobLeaseLostError()

    decision = decide_retry(
        error_code=error_code,
        attempt_count=job.attempt_count,
        max_attempts=job.max_attempts,
    )

    sanitized_message = (
        error_message[:_LEASE_MAX_MESSAGE] if error_message else None
    )

    if decision.retry and decision.delay_seconds is not None:
        validate_job_transition("processing", "retrying")
        new_status = "retrying"
        new_available_at = now + timedelta(
            seconds=decision.delay_seconds
        )
        new_failed_at = None
    else:
        validate_job_transition("processing", "failed")
        new_status = "failed"
        new_available_at = job.available_at
        new_failed_at = now

    job.status = new_status
    job.available_at = new_available_at
    job.failed_at = new_failed_at
    job.heartbeat_at = None
    job.worker_id = None
    job.last_error_code = (
        "max_attempts_exceeded"
        if decision.error_code == "max_attempts_exceeded"
        else error_code
    )
    job.last_error_message = sanitized_message
    job.updated_at = now
    db.commit()


def cancel_job(
    db: Session, *, job_id: uuid.UUID, user_id: uuid.UUID | None
) -> Job:
    now = datetime.now(UTC)

    stmt = select(Job).where(Job.id == job_id).with_for_update()
    if user_id is not None:
        stmt = stmt.where(Job.user_id == user_id)
    job = db.execute(stmt).scalar_one_or_none()

    if job is None:
        db.rollback()
        raise JobNotFoundError()

    if job.status not in ("queued", "retrying"):
        db.rollback()
        raise InvalidJobStateError(
            f"cannot cancel job in status {job.status!r}"
        )

    validate_job_transition(job.status, "cancelled")

    result = cast(
        CursorResult,
        db.execute(
            update(Job)
            .where(
                Job.id == job_id,
                Job.status.in_(("queued", "retrying")),
            )
            .values(status="cancelled", updated_at=now)
        ),
    )
    if result.rowcount == 0:
        db.rollback()
        raise InvalidJobStateError("job is no longer cancellable")
    db.commit()
    db.refresh(job)
    return job


def reclaim_stale_jobs(db: Session, *, limit: int = 100) -> int:
    settings = get_settings()
    now = datetime.now(UTC)
    cutoff = now - timedelta(
        seconds=settings.job_processing_timeout_seconds
    )

    try:
        stmt = (
            select(Job)
            .where(
                Job.status == "processing",
                Job.heartbeat_at.is_not(None),
                Job.heartbeat_at < cutoff,
            )
            .order_by(Job.heartbeat_at.asc())
            .limit(limit)
            .with_for_update(skip_locked=True)
        )
        jobs = list(db.execute(stmt).scalars().all())
        if not jobs:
            db.rollback()
            return 0

        for job in jobs:
            job.worker_id = None
            job.heartbeat_at = None
            job.updated_at = now
            if job.attempt_count < job.max_attempts:
                validate_job_transition(job.status, "retrying")
                job.status = "retrying"
                job.available_at = now
                job.last_error_code = "lease_expired"
                job.last_error_message = (
                    "worker heartbeat expired before completion"
                )
            else:
                validate_job_transition(job.status, "failed")
                job.status = "failed"
                job.failed_at = now
                job.last_error_code = "lease_expired"
                job.last_error_message = (
                    "worker heartbeat expired and no attempts remain"
                )
        db.commit()
        return len(jobs)
    except Exception:
        db.rollback()
        raise

def _evidence_role(
    matched_requirement_keys: list[str],
    required_keys: set[str],
) -> str:
    if any(k in required_keys for k in matched_requirement_keys):
        return "primary"
    return "support"


def run_section_research_persisted(
    db: Session,
    *,
    user_id: uuid.UUID,
    report_id: uuid.UUID,
    section_id: uuid.UUID,
    force: bool = False,
    commit: bool = True,
) -> PersistedSectionResearchResult:
    settings = get_settings()
    if not settings.section_research_enabled:
        raise SectionResearchConfigurationInvalid(
            "section research is disabled"
        )

    section = db.get(ManuscriptSection, section_id)
    if section is None:
        raise SectionResearchNotFound()
    version = db.get(ManuscriptVersion, section.manuscript_version_id)
    if version is None:
        raise SectionResearchNotFound()
    manuscript = db.get(Manuscript, version.manuscript_id)
    if manuscript is None or manuscript.report_id != report_id:
        raise SectionResearchNotFound()
    if manuscript.current_version_id != version.id:
        raise SectionResearchInvalid(
            "section is not in the manuscript's current version"
        )
    project_id = manuscript.report_id  # placeholder; corrected below
    # Resolve real project_id via reports
    from app.db.models.report import Report
    report = db.get(Report, report_id)
    if report is None or report.project_id is None:
        raise SectionResearchInvalid("report has no project")
    project_id = report.project_id

    try:
        _lock_section(db, section_id=section_id)

        context_hash, retrieval_config = _prepare_context_hash(
            db,
            section=section,
            version=version,
            manuscript=manuscript,
            project_id=project_id,
            report_id=report_id,
            user_id=user_id,
        )

        existing = _completed_run(
            db,
            report_id=report_id,
            section_id=section_id,
            context_hash=context_hash,
        )
        if existing is not None:
            now = datetime.now(UTC)
            skipped = SectionResearchRun(
                project_id=project_id,
                report_id=report_id,
                manuscript_id=manuscript.id,
                manuscript_version_id=version.id,
                manuscript_section_id=section.id,
                context_hash=context_hash,
                contract_hash=existing.contract_hash,
                m28_version=SECTION_RESEARCH_VERSION,
                status="skipped",
                primary_query=existing.primary_query,
                query_plan=existing.query_plan,
                retrieval_config=retrieval_config,
                retrieval_run_id=existing.retrieval_run_id,
                candidate_count=existing.candidate_count,
                selected_evidence_count=existing.selected_evidence_count,
                coverage_score=existing.coverage_score,
                coverage_state=existing.coverage_state,
                coverage_summary=existing.coverage_summary,
                warnings=["reused_previous_completed_run"],
                duration_ms=0,
                completed_at=now,
            )
            db.add(skipped)
            if commit:
                db.commit()
                db.refresh(skipped)
            else:
                db.flush()
            return PersistedSectionResearchResult(
                run=skipped, skipped=True,
            )

        result = engine_svc.run_section_research(
            db,
            user_id=user_id,
            report_id=report_id,
            section_id=section_id,
        )

        required_keys = {
            r["requirement_key"]
            for r in (
                section.research_contract or {}
            ).get("evidence_requirements") or []
            if r.get("required")
        }

        now = datetime.now(UTC)
        run = SectionResearchRun(
            project_id=project_id,
            report_id=report_id,
            manuscript_id=manuscript.id,
            manuscript_version_id=version.id,
            manuscript_section_id=section.id,
            context_hash=result.context_hash,
            contract_hash=result.contract_hash,
            m28_version=SECTION_RESEARCH_VERSION,
            status=result.status,
            primary_query=result.primary_query,
            query_plan=result.query_plan.as_dict(),
            retrieval_config=retrieval_config,
            retrieval_run_id=result.retrieval_run_id,
            candidate_count=int(result.candidates_considered),
            selected_evidence_count=len(result.evidence),
            coverage_score=float(result.coverage.coverage_score),
            coverage_state=result.coverage.coverage_state,
            coverage_summary=result.coverage.as_dict(),
            warnings=list(result.warnings),
            error_message=result.error_message,
            duration_ms=int(result.duration_ms),
            completed_at=now,
        )
        db.add(run)
        db.flush()

        for idx, ev in enumerate(result.evidence, start=1):
            db.add(SectionResearchEvidence(
                research_run_id=run.id,
                project_id=project_id,
                source_id=ev.source_id,
                chunk_id=ev.chunk_id,
                rank=idx,
                final_score=float(ev.final_score),
                retrieval_stage=ev.retrieval_stage,
                retrieval_reason=list(ev.retrieval_reason),
                evidence_role=_evidence_role(
                    list(ev.matched_requirement_keys), required_keys,
                ),
                matched_requirement_keys=list(
                    ev.matched_requirement_keys
                ),
                citation_label=ev.citation_label,
                excerpt=ev.excerpt,
                evidence_meta=None,
            ))

        if commit:
            db.commit()
            db.refresh(run)
        else:
            db.flush()

        return PersistedSectionResearchResult(run=run, skipped=False)

    except Exception:
        logger.exception("section_research_persist_failed")
        if commit:
            db.rollback()
        raise

def enqueue_section_research(
    db: Session,
    *,
    report_id: uuid.UUID,
    section_id: uuid.UUID,
    user_id: uuid.UUID,
    force: bool = False,
) -> uuid.UUID:
    """Enqueue a section_research job.

    Raises:
        SectionResearchNotFound — if the chain does not resolve.
        Any job_service error is re-raised.
    """
    report = db.get(Report, report_id)
    if report is None or report.project_id is None:
        raise SectionResearchNotFound()

    section = db.get(ManuscriptSection, section_id)
    if section is None:
        raise SectionResearchNotFound()
    version = db.get(ManuscriptVersion, section.manuscript_version_id)
    if version is None:
        raise SectionResearchNotFound()
    manuscript = db.get(Manuscript, version.manuscript_id)
    if manuscript is None or manuscript.report_id != report_id:
        raise SectionResearchNotFound()
    if manuscript.current_version_id != version.id:
        raise SectionResearchNotFound()

    vc = validate_contract(
        contract=section.research_contract or {},
        section_key=section.section_key,
    )

    settings = get_settings()
    query_plan = q_svc.build_query_plan(
        research_objective=vc.research_objective,
        retrieval_intents=list(vc.retrieval_intents),
        max_queries=int(settings.section_research_max_queries),
    )
    eff_filters, _w = filt_svc.build_effective_filters(
        source_constraints=vc.source_constraints,
    )

    profile = get_current_embedding_profile()
    provider = build_embedding_provider(profile)

    m26_ctx = ret_svc.resolve_section_retrieval_context(
        db,
        section=section,
        project_id=report.project_id,
        report_id=report_id,
        user_id=user_id,
        query=query_plan.primary,
        provider=provider,
        profile=profile,
        mode="lexical",
        algorithm="ann",
        max_results=None,
        min_year=eff_filters.min_year,
        max_year=eff_filters.max_year,
        required_full_text=eff_filters.required_full_text,
        allowed_source_ids=None,
    )

    context_hash = ctx_svc.compute_section_research_context_hash(
        project_id=report.project_id,
        report_id=report_id,
        manuscript_version_id=version.id,
        manuscript_section_id=section.id,
        contract_hash=vc.contract_hash,
        profile=profile,
        mode=m26_ctx.mode,
        algorithm=m26_ctx.algorithm,
        effective_filters=eff_filters.as_dict(),
        effective_max_results=m26_ctx.config.max_results,
        candidate_limit=m26_ctx.config.candidate_limit,
    )

    payload = {
        "report_id": str(report_id),
        "section_id": str(section_id),
        "user_id": str(user_id),
        "context_hash": context_hash,
        "contract_hash": vc.contract_hash,
        "m28_version": SECTION_RESEARCH_VERSION,
        "force": bool(force),
    }

    enqueued = jobs_service.enqueue_job(
        db,
        job_type="section_research",
        payload=payload,
        idempotency_key=(
            f"section_research:{section_id}:"
            f"{context_hash}:{SECTION_RESEARCH_VERSION}"
        ),
        report_id=report_id,
        user_id=user_id,
        max_attempts=int(settings.section_research_job_max_attempts),
    )
    return enqueued.job.id

def _load_manuscript_chain(
    db: Session, *, user_id: uuid.UUID, report_id: uuid.UUID,
    section_id: uuid.UUID,
) -> tuple[
    ManuscriptSection, ManuscriptVersion, Manuscript, uuid.UUID,
]:
    report = get_report(db, user_id=user_id, report_id=report_id)
    if report.project_id is None:
        raise SectionResearchInvalid("report has no project")

    section = db.get(ManuscriptSection, section_id)
    if section is None:
        raise SectionResearchNotFound()

    version = db.get(ManuscriptVersion, section.manuscript_version_id)
    if version is None:
        raise SectionResearchNotFound()

    manuscript = db.get(Manuscript, version.manuscript_id)
    if manuscript is None or manuscript.report_id != report_id:
        # Do not leak existence of foreign sections.
        raise SectionResearchNotFound()

    if manuscript.current_version_id != version.id:
        raise SectionResearchInvalid(
            "section is not in the manuscript's current version"
        )
    if version.status != "ready":
        raise SectionResearchInvalid("manuscript version is not ready")

    return section, version, manuscript, report.project_id


def _load_sources(
    db: Session, *, source_ids: set[uuid.UUID],
) -> dict[uuid.UUID, Source]:
    if not source_ids:
        return {}
    rows = db.execute(
        select(Source).where(Source.id.in_(list(source_ids)))
    ).scalars().all()
    return {s.id: s for s in rows}


def _load_literature(
    db: Session, *, project_id: uuid.UUID,
    source_ids: set[uuid.UUID],
) -> dict[uuid.UUID, ProjectLiteratureItem]:
    if not source_ids:
        return {}
    rows = db.execute(
        select(ProjectLiteratureItem).where(
            ProjectLiteratureItem.project_id == project_id,
            ProjectLiteratureItem.source_id.in_(list(source_ids)),
        )
    ).scalars().all()
    return {r.source_id: r for r in rows}


def _load_chunk_contents(
    db: Session, *, chunk_ids: set[uuid.UUID],
) -> dict[uuid.UUID, str]:
    if not chunk_ids:
        return {}
    rows = db.execute(
        select(Chunk.id, Chunk.content).where(
            Chunk.id.in_(list(chunk_ids))
        )
    ).tuples().all()
    return dict(rows)

def run_section_research(
    db: Session,
    *,
    user_id: uuid.UUID,
    report_id: uuid.UUID,
    section_id: uuid.UUID,
    provider=None,
    profile: EmbeddingProfile | None = None,
) -> SectionResearchResult:
    started = time.perf_counter()
    settings = get_settings()

    if not settings.section_research_enabled:
        raise SectionResearchConfigurationInvalid(
            "section research is disabled"
        )

    section, version, manuscript, project_id = _load_manuscript_chain(
        db,
        user_id=user_id,
        report_id=report_id,
        section_id=section_id,
    )
    project = get_project(db, user_id=user_id, project_id=project_id)

    # 1) Contract validation (fail-closed)
    vc: ValidatedContract = validate_contract(
        contract=section.research_contract or {},
        section_key=section.section_key,
    )

    # 2) Query plan
    query_plan = q_svc.build_query_plan(
        research_objective=vc.research_objective,
        retrieval_intents=list(vc.retrieval_intents),
        max_queries=int(settings.section_research_max_queries),
    )

    # 3) Effective filters (deterministic)
    eff_filters, filter_warnings = filt_svc.build_effective_filters(
        source_constraints=vc.source_constraints,
        reference_year=RESEARCH_PLAN_REFERENCE_YEAR,
    )

    # 4) Provider/profile
    if profile is None:
        profile = get_current_embedding_profile()
    if provider is None:
        provider = build_embedding_provider(profile)

    # 5) Resolve M26 context (drives effective config + M26 context_hash)
    m26_ctx = ret_svc.resolve_section_retrieval_context(
        db,
        section=section,
        project_id=project.id,
        report_id=report_id,
        user_id=user_id,
        query=query_plan.primary,
        provider=provider,
        profile=profile,
        mode="lexical",
        algorithm="ann",
        max_results=None,
        min_year=eff_filters.min_year,
        max_year=eff_filters.max_year,
        required_full_text=eff_filters.required_full_text,
        allowed_source_ids=None,
    )

    # 6) M28 context hash (deterministic, independent from M26 hash)
    context_hash = ctx_svc.compute_section_research_context_hash(
        project_id=project.id,
        report_id=report_id,
        manuscript_version_id=version.id,
        manuscript_section_id=section.id,
        contract_hash=vc.contract_hash,
        profile=profile,
        mode=m26_ctx.mode,
        algorithm=m26_ctx.algorithm,
        effective_filters=eff_filters.as_dict(),
        effective_max_results=m26_ctx.config.max_results,
        candidate_limit=m26_ctx.config.candidate_limit,
    )

    # 7) Run M26
    warnings: list[str] = list(filter_warnings)
    outcome = None
    candidates: list[dict] = []
    try:
        bundle = ret_svc.execute_section_retrieval(
            db,
            section=section,
            project_id=project.id,
            report_id=report_id,
            user_id=user_id,
            query=query_plan.primary,
            provider=provider,
            profile=profile,
            context=m26_ctx,
        )
        outcome = bundle.outcome
        if outcome is not None:
            candidates = list(outcome.results or [])
        warnings.extend(bundle.warnings)
    except Exception as exc:
        logger.exception("section_research_m26_failed")
        duration_ms = int((time.perf_counter() - started) * 1000)
        return SectionResearchResult(
            section_id=section.id,
            manuscript_version_id=version.id,
            manuscript_id=manuscript.id,
            report_id=report_id,
            project_id=project.id,
            section_key=section.section_key,
            contract_hash=vc.contract_hash,
            context_hash=context_hash,
            query_plan=query_plan,
            primary_query=query_plan.primary,
            retrieval_run_id=None,
            candidates_considered=0,
            evidence=[],
            per_requirement=[],
            coverage=cov_svc.SectionCoverage(
                coverage_version="m28-coverage-v1",
                coverage_target=vc.coverage_target,
                coverage_score=0.0,
                coverage_state="insufficient",
            ),
            warnings=warnings,
            status="failed",
            error_message=str(exc)[:500],
            duration_ms=duration_ms,
        )

    # 8) Load sources + literature + chunks
    source_ids: set[uuid.UUID] = set()
    chunk_ids: set[uuid.UUID] = set()
    for c in candidates:
        sid = c.get("source_id")
        if sid is not None:
            source_ids.add(sid)
        cid = c.get("chunk_id")
        if cid is not None:
            chunk_ids.add(cid)

    sources_by_id = _load_sources(db, source_ids=source_ids)
    literature_by_source_id = _load_literature(
        db, project_id=project.id, source_ids=source_ids,
    )
    chunk_contents = _load_chunk_contents(db, chunk_ids=chunk_ids)

    # 9) Evidence selection
    selection = ev_svc.select_evidence(
        candidates=candidates,
        requirements=vc.evidence_requirements,
        sources_by_id=sources_by_id,
        literature_by_source_id=literature_by_source_id,
        chunk_contents=chunk_contents,
        excerpt_chars=int(settings.section_research_excerpt_chars),
        max_evidence=int(settings.section_research_max_evidence),
        per_source_cap=ev_svc.DEFAULT_PER_SOURCE_CAP,
        reference_year=RESEARCH_PLAN_REFERENCE_YEAR,
    )
    warnings.extend(selection.warnings)

    # 10) Coverage
    coverage = cov_svc.evaluate_section_coverage(
        requirements=vc.evidence_requirements,
        matches=selection.matches,
        coverage_target=vc.coverage_target,
        covered_threshold=float(
            settings.section_research_coverage_covered_threshold
        ),
        partial_threshold=float(
            settings.section_research_coverage_partial_threshold
        ),
    )
    warnings.extend(coverage.warnings)

    duration_ms = int((time.perf_counter() - started) * 1000)
    return SectionResearchResult(
        section_id=section.id,
        manuscript_version_id=version.id,
        manuscript_id=manuscript.id,
        report_id=report_id,
        project_id=project.id,
        section_key=section.section_key,
        contract_hash=vc.contract_hash,
        context_hash=context_hash,
        query_plan=query_plan,
        primary_query=query_plan.primary,
        retrieval_run_id=outcome.run.id if outcome else None,
        candidates_considered=len(candidates),
        evidence=selection.selected,
        per_requirement=coverage.per_requirement,
        coverage=coverage,
        warnings=warnings,
        status="completed",
        error_message=None,
        duration_ms=duration_ms,
    )


__all__ = ["SectionResearchResult", "run_section_research"]

"""Job handler registry."""
from __future__ import annotations

import logging
from collections.abc import Callable

from app.domain.jobs import JobContext

logger = logging.getLogger(__name__)

JobHandler = Callable[[JobContext], object]
_REGISTRY: dict[str, JobHandler] = {}


class HandlerAlreadyRegistered(Exception):
    pass


class HandlerNotFound(Exception):
    pass


def register_handler(
    job_type: str, handler: JobHandler, *, override: bool = False
) -> None:
    if not job_type:
        raise ValueError("job_type must be non-empty")
    if job_type in _REGISTRY and not override:
        raise HandlerAlreadyRegistered(job_type)
    _REGISTRY[job_type] = handler


def get_handler(job_type: str) -> JobHandler:
    handler = _REGISTRY.get(job_type)
    if handler is None:
        raise HandlerNotFound(job_type)
    return handler


def clear_registry() -> None:
    _REGISTRY.clear()


def registered_types() -> tuple[str, ...]:
    return tuple(sorted(_REGISTRY.keys()))


__all__ = [
    "HandlerAlreadyRegistered",
    "HandlerNotFound",
    "JobHandler",
    "clear_registry",
    "get_handler",
    "register_handler",
    "registered_types",
]

"""M28 — Worker handler for section_research jobs (stale-safe)."""
from __future__ import annotations

import logging

from app.db.models.manuscript import (

def _parse_uuid(value) -> uuid.UUID | None:
    if value is None:
        return None
    try:
        return uuid.UUID(str(value))
    except (ValueError, AttributeError, TypeError):
        return None


def handle_section_research(ctx) -> dict:
    payload = ctx.payload or {}
    report_id = _parse_uuid(payload.get("report_id"))
    section_id = _parse_uuid(payload.get("section_id"))
    if report_id is None or section_id is None:
        raise ValueError("section_research payload missing ids")

    user_id = _parse_uuid(payload.get("user_id"))
    payload_context_hash = payload.get("context_hash")
    payload_contract_hash = payload.get("contract_hash")
    payload_m28_version = payload.get("m28_version")
    force = bool(payload.get("force", False))

    with SessionLocal() as db:
        report = db.get(Report, report_id)
        if report is None or report.project_id is None:
            return {
                "status": "skipped", "skipped": True,
                "reason": "report_has_no_project",
            }
        effective_user_id = user_id or report.user_id

        section = db.get(ManuscriptSection, section_id)
        if section is None:
            return {
                "status": "skipped", "skipped": True,
                "reason": "section_not_found",
            }
        version = db.get(ManuscriptVersion, section.manuscript_version_id)
        if version is None:
            return {
                "status": "skipped", "skipped": True,
                "reason": "version_not_found",
            }
        manuscript = db.get(Manuscript, version.manuscript_id)
        if manuscript is None or manuscript.report_id != report_id:
            return {
                "status": "skipped", "skipped": True,
                "reason": "manuscript_mismatch",
            }
        if manuscript.current_version_id != version.id:
            return {
                "status": "skipped", "skipped": True,
                "reason": "stale_version",
            }

        # Recompute current context_hash from the real, current contract
        try:
            vc = validate_contract(
                contract=section.research_contract or {},
                section_key=section.section_key,
            )
        except Exception:
            return {
                "status": "skipped", "skipped": True,
                "reason": "contract_invalid",
            }

        settings_queries = int(
            __import__(
                "app.core.config", fromlist=["get_settings"],
            ).get_settings().section_research_max_queries
        )
        query_plan = q_svc.build_query_plan(
            research_objective=vc.research_objective,
            retrieval_intents=list(vc.retrieval_intents),
            max_queries=settings_queries,
        )
        eff_filters, _w = filt_svc.build_effective_filters(
            source_constraints=vc.source_constraints,
        )
        profile = get_current_embedding_profile()
        provider = build_embedding_provider(profile)
        m26_ctx = ret_svc.resolve_section_retrieval_context(
            db,
            section=section,
            project_id=report.project_id,
            report_id=report_id,
            user_id=effective_user_id,
            query=query_plan.primary,
            provider=provider,
            profile=profile,
            mode="lexical",
            algorithm="ann",
            max_results=None,
            min_year=eff_filters.min_year,
            max_year=eff_filters.max_year,
            required_full_text=eff_filters.required_full_text,
            allowed_source_ids=None,
        )
        current_context_hash = (
            ctx_svc.compute_section_research_context_hash(
                project_id=report.project_id,
                report_id=report_id,
                manuscript_version_id=version.id,
                manuscript_section_id=section.id,
                contract_hash=vc.contract_hash,
                profile=profile,
                mode=m26_ctx.mode,
                algorithm=m26_ctx.algorithm,
                effective_filters=eff_filters.as_dict(),
                effective_max_results=m26_ctx.config.max_results,
                candidate_limit=m26_ctx.config.candidate_limit,
            )
        )

        stale_reason: str | None = None
        if payload_context_hash and payload_context_hash != (
            current_context_hash
        ):
            stale_reason = "stale_context"
        elif payload_contract_hash and payload_contract_hash != (
            vc.contract_hash
        ):
            stale_reason = "stale_contract"
        elif payload_m28_version and payload_m28_version != (
            SECTION_RESEARCH_VERSION
        ):
            stale_reason = "stale_version"

        if stale_reason is not None:
            logger.info(
                "section_research_stale report_id=%s section_id=%s "
                "reason=%s",
                report_id, section_id, stale_reason,
            )
            return {
                "status": "skipped", "skipped": True,
                "reason": stale_reason,
                "payload_context_hash": payload_context_hash,
                "current_context_hash": current_context_hash,
            }

        persisted = persist_svc.run_section_research_persisted(
            db,
            user_id=effective_user_id,
            report_id=report_id,
            section_id=section_id,
            force=force,
            commit=True,
        )
        return {
